def confirm(day: datetime.date | None = None) -> tuple[str, str]:
    """Return (status, detail) where status is 'sent', 'failed' or 'unknown'.

    'unknown' is reserved for genuinely undecidable cases — Outlook unreachable —
    so it stops being the catch-all it was when the log was parsed for prose.
    """
    day = day or datetime.date.today()
    if not report_html(day):
        return "failed", (f"No digest was built for {day:%Y-%m-%d} "
                          f"(report/{day:%m-%d-%Y}/ has no ranked.html).")

    subject = expected_subject(day)
    if not subject:
        return "unknown", ("The digest HTML exists but its subject could not be "
                           "parsed, so delivery could not be verified.")

    found = in_sent_items(subject, day)
    if found is True:
        return "sent", f"'{subject}' confirmed in Outlook Sent Items."
    if found is False:
        return "failed", (f"'{subject}' was built but is NOT in Outlook Sent "
                          f"Items — check the Outbox; subscribers likely got "
                          f"nothing.")

    # Outlook unreachable: the sender's own marker is weaker evidence (it records
    # that Send() was confirmed earlier, not that the message is there now), but
    # it beats reporting a delivered digest as undecidable.
    marker = read_marker(day)
    if marker:
        return "sent", (f"'{subject}' recorded as sent to "
                        f"{marker.get('recipients', '?')} recipients at "
                        f"{marker.get('when', '?')} (Outlook unreachable, so this "
                        f"is the sender's own record rather than Sent Items).")
    return "unknown", (f"'{subject}' was built, but Outlook could not be reached "
                       f"to confirm delivery. Check Sent Items manually.")

def already_sent(day: datetime.date | None = None) -> bool:
    """Cheap, Outlook-free 'did today's push already go out?' for catch-up runs.

    Only ever answers True on positive evidence, so a false negative costs a
    duplicate build (annoying) while a false positive would cost the whole
    digest (silent miss) — the asymmetry decides which way this leans.
    """
    if read_marker(day):
        return True
    subject = expected_subject(day)
    return bool(subject) and in_sent_items(subject, day) is True

def _parse_date(raw: str) -> "datetime | None":
    raw = raw.strip()
    if not raw:
        return None

    # RFC 2822 (standard RSS pubDate)
    try:
        return parsedate_to_datetime(raw).astimezone(timezone.utc)
    except Exception:
        pass

    # ISO 8601 variants (incl. fractional seconds, e.g. Reuters' "...:40.147Z")
    for fmt in ("%Y-%m-%dT%H:%M:%S.%f%z", "%Y-%m-%dT%H:%M:%S.%fZ",
                "%Y-%m-%dT%H:%M:%S%z", "%Y-%m-%dT%H:%M:%SZ", "%Y-%m-%d"):
        try:
            dt = datetime.strptime(raw, fmt)
            if dt.tzinfo is None:
                dt = dt.replace(tzinfo=timezone.utc)
            return dt.astimezone(timezone.utc)
        except ValueError:
            pass

    # Human-readable: "June 1, 2026" / "Jun 1, 2026" / "1 June 2026"
    for fmt in ("%B %d, %Y", "%b %d, %Y", "%d %B %Y", "%d %b %Y"):
        try:
            return datetime.strptime(raw, fmt).replace(tzinfo=timezone.utc)
        except ValueError:
            pass

    return None


def _in_window(dt: "datetime | None", cutoff: datetime) -> bool:
    if dt is None:
        return False
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt >= cutoff

# Below this many characters we treat the page as a paywall preview, not an article.
TRUNCATED_BELOW = 600
# Cap stored text so one giant analyst post doesn't dominate the output file.
DEFAULT_MAX_CHARS = 8000


def extract_text(html: str) -> str:
    """Heuristic main-body extraction: drop chrome, then take the container with
    the most substantial <p> text. No third-party readability lib required."""
    soup = BeautifulSoup(html, "html.parser")
    for tag in soup(["script", "style", "nav", "header", "footer", "aside",
                     "form", "figure", "noscript", "button"]):
        tag.decompose()

    def collect(container) -> str:
        parts = []
        for p in container.find_all("p", recursive=True):
            txt = p.get_text(" ", strip=True)
            if len(txt) > 40 and not _BOILERPLATE.search(txt):
                parts.append(txt)
        return "\n".join(parts)

    best = ""
    for container in soup.find_all(["article", "main", "div", "section"]):
        txt = collect(container)
        if len(txt) > len(best):
            best = txt

    if not best:                       # fall back to every <p> on the page
        best = collect(soup)

    return best.strip()


def fetch_one(url: str, max_chars: int) -> dict:
    resp, reason = _fetch(url)
    if resp is None:
        return {"url": url, "status": "BLOCKED", "chars": 0,
                "text": f"(could not fetch — {reason}; fall back to snippet)"}

    text = extract_text(resp.text)
    if len(text) < TRUNCATED_BELOW:
        return {"url": url, "status": "TRUNCATED", "chars": len(text),
                "text": (text or "(no article body found)") +
                        "\n(only a preview was available — fall back to snippet)"}

    return {"url": url, "status": "OK", "chars": len(text),
            "text": text[:max_chars]}

def crop_to_fill(src: Path, w: int = THUMB_W, h: int = THUMB_H,
                  scale: int = THUMB_SCALE) -> Path:
    """Center-crop src to exactly (w*scale) x (h*scale) pixels (aspect-fill, like CSS
    object-fit:cover) and cache the result next to it. Rendered at `scale`x the display
    size so it stays sharp under Windows/Outlook display scaling — the <img> tag still
    declares width=w height=h, so this larger bitmap gets downsampled for display, not
    upscaled from a same-size original. Outlook desktop's Word rendering engine also
    ignores CSS object-fit, so pre-cropping the actual file (rather than cropping via
    CSS at display time) is the only way to get an undistorted thumbnail in every
    client, including Outlook."""
    px_w, px_h = w * scale, h * scale
    out = src.with_name(f"{src.stem}_crop{px_w}x{px_h}.jpg")
    if out.exists() and out.stat().st_mtime >= src.stat().st_mtime:
        return out
    with Image.open(src) as im:
        im = im.convert("RGB")
        fill_scale = max(px_w / im.width, px_h / im.height)
        new_w = max(px_w, round(im.width * fill_scale))
        new_h = max(px_h, round(im.height * fill_scale))
        im = im.resize((new_w, new_h), Image.LANCZOS)
        left, top = (new_w - px_w) // 2, (new_h - px_h) // 2
        im = im.crop((left, top, left + px_w, top + px_h))
        im.save(out, "JPEG", quality=88)
    return out

def attach_inline_images_map(mail, cid_map: dict) -> int:
    """Same as attach_inline_images, but takes an in-memory {cid: path} map directly —
    used by the per-subscriber daily sender, which builds one cid_map and reuses it
    across many individual mail items instead of round-tripping through a sidecar file."""
    n = 0
    for cid, path in cid_map.items():
        p = Path(path)
        if not p.exists():
            print(f"[warn] inline image missing, skipping: {p}")
            continue
        try:
            att = mail.Attachments.Add(str(p.resolve()))
            pa = att.PropertyAccessor
            pa.SetProperty(PR_ATTACH_CONTENT_ID, cid)
            try:
                pa.SetProperty(PR_ATTACHMENT_HIDDEN, True)
            except Exception:
                pass          # not fatal — image still renders, just also lists as a file
            n += 1
        except Exception as exc:
            print(f"[warn] could not attach inline image {p.name}: {exc}")
    return n

def _sender_smtp(item) -> str:
    """Real SMTP address for the sender. Exchange-internal senders report an
    X.500 directory name (e.g. "/O=EXCHANGELABS/OU=.../CN=...") via
    SenderEmailAddress, not a usable email — GetExchangeUser().PrimarySmtpAddress
    is needed for those. External/SMTP senders already have a plain address."""
    try:
        if getattr(item, "SenderEmailType", "") == EXCHANGE_ADDRESS_TYPE:
            return item.Sender.GetExchangeUser().PrimarySmtpAddress
    except Exception:
        pass
    return item.SenderEmailAddress
