def only_recipients(cli_value: str = "") -> list:
    """Addresses this send is clamped to (lowercased); empty means no clamp."""
    raw = cli_value or os.environ.get(ONLY_ENV, "")
    return [e.strip().lower() for e in re.split(r"[;,]", raw) if e.strip()]
