def _sender_smtp(item) -> str:
    """Real SMTP address for the sender. Exchange-internal senders report an
    X.500 directory name (e.g. "/O=EXCHANGELABS/OU=.../CN=...") via
    SenderEmailAddress, not a usable email — GetExchangeUser().PrimarySmtpAddress
    is needed for those. External/SMTP senders already have a plain address."""
    try:
        if getattr(item, "SenderEmailType", "") == EXCHANGE_ADDRESS_TYPE:
            return item.Sender.GetExchangeUser().PrimarySmtpAddress
    except Exception:
        pass
    return item.SenderEmailAddress

def _classify_tag(subject: str) -> str:
    """Exact-tag subscribe/unsubscribe classification (the mailto: link path)."""
    s = (subject or "").upper()
    if UNSUBSCRIBE_TAG in s:
        # UNSUBSCRIBE_IC_NEWS contains "SUBSCRIBE_IC_NEWS" as a substring, so
        # check this first — a subject with both tags is treated as unsubscribe.
        return "unsubscribe"
    if SUBSCRIBE_TAG in s:
        return "subscribe"
    return ""

def save_subscribers(data: dict) -> None:
    """Atomic write (temp file + replace) so a crash or a concurrent reader never
    sees a half-written/corrupted subscribers.json."""
    tmp = SUBSCRIBERS_PATH.with_suffix(".json.tmp")
    tmp.write_text(json.dumps(data, indent=2), encoding="utf-8")
    os.replace(tmp, SUBSCRIBERS_PATH)
